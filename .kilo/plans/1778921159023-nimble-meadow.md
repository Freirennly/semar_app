## 3. Route Laravel yang Perlu Ditambahkan

```php
// Dashboard Admin Enhancement
Route::middleware('role:admin')->prefix('admin')->name('admin.')->group(function () {
    // Dashboard overview
    Route::get('dashboard', [AdminDashboardController::class, 'index'])->name('dashboard');
    
    // User Management Enhancement
    Route::post('users/{user}/activate', [UserController::class, 'activate'])->name('users.activate');
    Route::post('users/{user}/deactivate', [UserController::class, 'deactivate'])->name('users.deactivate');
    Route::post('users/{user}/reset-password', [UserController::class, 'resetPassword'])->name('users.reset-password');
    
    // Proposal Monitoring
    Route::get('proposals', [AdminProposalController::class, 'index'])->name('proposals.index');
    Route::get('proposals/{submission}', [AdminProposalController::class, 'show'])->name('proposals.show');
    
    // Reviewer Monitoring  
    Route::get('reviewers', [AdminReviewerController::class, 'index'])->name('reviewers.index');
    Route::get('reviewers/{user}', [AdminReviewerController::class, 'show'])->name('reviewers.show');
    
    // Audit Log
    Route::get('audit-logs', [AdminAuditController::class, 'index'])->name('audit-logs.index');
    
    // Export/Backup
    Route::get('export/proposals', [AdminExportController::class, 'exportProposals'])->name('export.proposals');
    Route::get('export/users', [AdminExportController::class, 'exportUsers'])->name('export.users');
});
```

```php
$approvedCount = Submission::where('status', SubmissionStatus::APPROVED)->count();
$rejectedCount = Submission::where('status', SubmissionStatus::DISUBMITTED)->count();

```

**Per Status Distribution:**
```php
$statusCounts = Submission::select('status', DB::raw('count(*) as count'))
```
